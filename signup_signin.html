import { configureSupabase, getSupabase } from "./assets/js/supabaseClient.js"; import { SUPABASE_CONFIG } from "./assets/js/config.js"; import { loadAuthGate } from "./assets/js/loadAuthGate.js"; import { initAuthGate } from "./assets/js/authGate.js"; configureSupabase(SUPABASE_CONFIG); await loadAuthGate(); const sb = getSupabase(); function repoBasePath() { // For GitHub Pages: //... // Example: /csedmaecapstone/signup_signin.html -> "/csedmaecapstone/" const parts = window.location.pathname.split("/").filter(Boolean); const repo = parts[0] || ""; return `/${repo}/`; } function urlFromRepoRoot(file) { return `${window.location.origin}${repoBasePath()}${file}`; } function resolveReturnTo() { const params = new URLSearchParams(window.location.search); const raw = params.get("returnTo"); // If missing, go to hub if (!raw) return urlFromRepoRoot("index.html"); // returnTo is expected to be an encoded path like: // %2Fcsedmaecapstone%2Fself_intro.html // We only allow same-origin paths to prevent open redirects. let decoded = ""; try { decoded = decodeURIComponent(raw); } catch { return urlFromRepoRoot("index.html"); } // Must start with "/" and must stay within this repo base path const base = repoBasePath(); // e.g., "/csedmaecapstone/" if (!decoded.startsWith(base)) return urlFromRepoRoot("index.html"); return `${window.location.origin}${decoded}`; } const targetUrl = resolveReturnTo(); // If already signed in, skip UI and go where they wanted { const { data, error } = await sb.auth.getSession(); if (error) console.error(error); if (data?.session) { window.location.replace(targetUrl); } else { await initAuthGate({ onAuthed: () => { window.location.replace(targetUrl); } }); } }
